# Landing parameters shared by every M2M_MODE.
# Per-mode values live in landing_sim.txt / landing_mocap.txt / landing_hw.txt.
# Units are in the name: _M metres, _S seconds, _MPS m/s, _MPS2 m/s^2, _RAD radians, _RADPS rad/s.
# The Lander refuses to start if a key is missing, unknown, duplicated, or in the wrong file.

# ======================================================================
# INSPEC: what INIT does if the pad is NOT seen at the start
# Not used once the landing has started (losing the pad mid-landing makes the drone climb, see OFF_CENTER_* below).
# ======================================================================
# 1 = INSPE mode: fly a circle looking for the pad. 0 = HOVERING mode: hover in place.
INSPEC_ENABLE=1
# Radius of the INSPE search circle [m].
INSPEC_CIRCLE_RADIUS_M=1
# Speed along the INSPE search circle [m/s].
INSPEC_CIRCLE_SPEED_MPS=0.3
# The INSPE circle radius grows by this after each full lap [m].
INSPEC_CIRCLE_RADIUS_GROWTH_M=0.5
# Absolute world altitude flown in INSPE and HOVERING [m].
INSPEC_ALTITUDE_M=6.0

# ======================================================================
# ALTITUDE SAFETY LIMITS
# ======================================================================
# When not deliberately descending, the drone climbs back if it gets closer to the pad than this [m].
ALT_MIN_ABOVE_PAD_M=1.0
# HOLD's climbs (pad lost, or too close to the pad) stop this far below ALT_MAX_ABOVE_PAD_M [m].
ALT_MAX_ABOVE_PAD_MARGIN_M=0.1

# ======================================================================
# CENTERED OVER THE PAD (checked on every new vision frame)
# Each frame compares the drone's horizontal distance to the pad centre with a radius that depends on height:
# the radius is *_MIN at the pad (height 0), grows linearly to *_MAX at CENTERED_CONE_HEIGHT_M, and stays *_MAX above it
# (a cone: the lower the drone, the more precisely it must be centred).
# ======================================================================
# Height above the pad at which the cone radii reach their *_MAX value [m].
CENTERED_CONE_HEIGHT_M=5.0
# Centred radius at the pad [m]. Distance under the cone radius = frame counts as CENTRED.
CENTERED_CONE_RADIUS_MIN_M=0.6
# Centred radius at/above the cone height [m].
CENTERED_CONE_RADIUS_MAX_M=1.5
# Same as CENTERED_CONE_RADIUS_MIN_M, but used instead of it while in COMP [m].
COMP_CENTERED_CONE_RADIUS_MIN_M=0.30
# Same as CENTERED_CONE_RADIUS_MAX_M, but used instead of it while in COMP [m].
COMP_CENTERED_CONE_RADIUS_MAX_M=1.0
# Off-centre radius at the pad [m]. Distance above the cone radius = frame counts as OFF-CENTRE.
OFF_CENTER_CONE_RADIUS_MIN_M=0.80
# Off-centre radius at/above the cone height [m].
OFF_CENTER_CONE_RADIUS_MAX_M=2.0
# Consecutive centred frames needed before HOLD may descend or move on to COMP.
CENTERED_FRAMES_REQUIRED=70
# Consecutive centred frames needed in COMP/LAND.
COMP_CENTERED_FRAMES_REQUIRED=20
# Consecutive off-centre frames after which the pad counts as LOST (no new vision frame at all = lost immediately).
OFF_CENTER_FRAMES_TO_LOSE_PAD=70

# ======================================================================
# HORIZONTAL (XY) CONTROLLER -- used in HOLD (after the slope) and in COMP only
# ======================================================================
# 0 = DIRECT: follow the filtered pad position. 1 = PID on the vision error.
XY_CONTROL_MODE=1
# PID proportional gain (mode 1).
XY_PID_KP=0.45
# PID integral gain (mode 1).
XY_PID_KI=0.002
# PID derivative gain (mode 1).
XY_PID_KD=0.0
# Upper limit of the PID integral term (mode 1).
XY_PID_INT_MAX=1
# Lower limit of the PID integral term (mode 1).
XY_PID_INT_MIN=-1
# Largest position correction the PID may command [m] (mode 1).
XY_PID_OUTPUT_MAX_M=1.5
# Smoothing of the pad position target, 0..1, higher = smoother/slower (mode 0).
XY_DIRECT_FILTER_ALPHA=0.5
# Speed limit on that target [m/s] (mode 0).
XY_DIRECT_MAX_SPEED_MPS=3.0
# Mode 0 only: command = target + gain * (target - drone), i.e. aim past the target by this fraction of the
#   current XY error so PX4 closes the gap faster (an extra P term). 0 = command the target itself.
XY_DIRECT_OVERSHOOT_GAIN=0.5

# ======================================================================
# PAD ESTIMATE FROM VISION
# ======================================================================
# Two consecutive vision frames further apart than this are not used to compute the pad velocity
#   (the estimate restarts from the new frame) [s].
PAD_EST_VEL_SKIP_IF_FRAME_GAP_ABOVE_S=0.5

# ======================================================================
# INIT: fly to the START POINT
# The start point is behind the pad and above it; the SLOPE then glides from it to HOLD_HEIGHT_M above the pad.
# ======================================================================
# Horizontal distance of the start point from the pad centre [m]. NOTE: the start point's height is
#   (highest pad height seen) + HOLD_HEIGHT_M + this same value, giving a 45-degree slope.
INIT_START_POINT_XY_DIST_M=2.5
# Drone counts as AT the start point when its X and Y errors to it are each under this [m].
INIT_START_POINT_XY_TOL_M=0.5
# Same, for height [m].
INIT_START_POINT_Z_TOL_M=0.5
# Drone must stay at the start point this long before the slope begins [s].
INIT_START_POINT_HOLD_S=2.0
# Control ticks INIT waits before choosing between the approach and INSPE/HOVERING.
INIT_WAIT_TICKS=20

# ======================================================================
# SLOPE: timed glide from the start point down to HOLD_HEIGHT_M above the pad
# ======================================================================
# How long the glide takes [s].
SLOPE_DURATION_S=8.0
# Integral gain removing the drone's trailing lag behind the moving target [1/s]. 0 = off.
SLOPE_XY_KI=0.3
# Limit on that integral correction [m].
SLOPE_XY_INT_MAX_M=1.0

# ======================================================================
# HOLD: hover above the pad (all heights are drone-to-pad)
# ======================================================================
# Target hover height [m]; the slope ends here.
HOLD_HEIGHT_M=1.5
# Higher than this: step down [m].
HOLD_DESCEND_ABOVE_M=1.8
# Lower than this: COMP may start [m]. Between the two: just hold.
HOLD_COMP_BELOW_M=1.2
# Setpoint step per control tick when descending [m].
HOLD_DESCEND_STEP_M=0.5
# Setpoint step per control tick when climbing [m].
HOLD_ASCEND_STEP_M=0.5
# HOLD only steps down while the drone's horizontal speed (X and Y each) is under this [m/s].
HOLD_DESCEND_ALLOWED_MAX_XY_SPEED_MPS=0.3
# COMP conditions must hold without a break this long before COMP starts [s].
HOLD_TO_COMP_WAIT_S=3.0

# ======================================================================
# COMP: follow the heaving pad at close range
# Heave margin: the peak UPWARD pad speed over the last COMP_HEAVE_PEAK_WINDOW_S seconds, added (as metres) to the COMP height
# and to the GO height gate, so both grow in rough seas. Each sum is then capped.
# ======================================================================
# Drone-to-pad height kept in COMP in a calm sea [m].
COMP_HEIGHT_BASE_M=0.2
# Window for the heave margin's peak upward speed [s].
COMP_HEAVE_PEAK_WINDOW_S=7.5
# COMP height = min(base + heave margin, this) [m]. Above it nothing else happens: the drone just holds at the cap.
COMP_HEIGHT_CAP_M=0.7
# Closer to the pad than this: stop following and back off upwards [m].
COMP_BACKOFF_BELOW_M=0.1
# Resume following once farther than this [m].
COMP_RESUME_ABOVE_M=0.15
# When the pad rises faster than this, also match its vertical speed [m/s].
COMP_VEL_MATCH_ABOVE_MPS=0.1
# How far ahead the heave prediction is used [s]; should match the heave estimator's prediction horizon.
COMP_LEAD_TIME_S=0.1
# COMP height command = drone Z + gain * (wanted COMP height - current gap).
COMP_HEIGHT_P_GAIN=3.0

# ======================================================================
# GO DECISION: commit to landing (COMP -> LAND)
# ======================================================================
# X and Y errors to the pad centre must each be under this [m].
GO_XY_ERR_MAX_M=0.1
# Drone-to-pad height must be under this in a calm sea [m].
GO_HEIGHT_BASE_M=0.3
# Height gate = min(base + heave margin, this) [m]. Keep consistent with COMP_HEIGHT_CAP_M, or GO is never given.
GO_HEIGHT_CAP_M=0.5
# All of the above must hold without a break this long [s].
GO_HOLD_S=5.0
# Majority vote over the raw GO decision in this window [s].
GO_VOTE_WINDOW_S=1.5
# Fraction (0..1] of the window that must be GO.
GO_VOTE_FRACTION=0.6
# A GO or NO-GO decision is kept at least this long [s].
GO_LATCH_S=0.25
# A fresh, valid sonar reading under this makes the sea-state GO true whatever the roll/pitch/heave limits say [m]
# (the drone is effectively down). The position and height checks of COMP -> LAND still apply.
GO_FORCE_IF_SONAR_BELOW_M=0.2

# ======================================================================
# LAND: committed descent after GO
# ======================================================================
# Descent speed [m/s].
LAND_DESCEND_SPEED_MPS=1.0
# 1 = disarm immediately at GO and skip the descent and touchdown detection.
LAND_DISARM_ON_GO=1

# ======================================================================
# LOGGING ONLY
# ======================================================================
# How far back [s] to take the prediction compared with the real values at touchdown in the CSV log.
LOG_PRED_LOOKBACK_S=0.5

# ======================================================================
# SIMPLE LAND: the LAND task's basic mode (task parameter 0 = no platform)
# Lands straight down on the spot where the task started, without the pad or vision. Heights are world Z.
# ======================================================================
# Horizontal velocity command = gain * distance to the landing spot [1/s].
SIMPLE_LAND_XY_GAIN=1.0
# Descends only while within this horizontal distance of the spot (X and Y each) [m].
SIMPLE_LAND_XY_TOL_M=0.25
# Descent speed while higher than SIMPLE_LAND_DESC_SPEED_MAX_ABOVE_M [m/s].
SIMPLE_LAND_DESC_SPEED_MAX_MPS=0.3
# Above this height (world Z) the max descent speed is used [m].
SIMPLE_LAND_DESC_SPEED_MAX_ABOVE_M=1.6
# Descent speed while lower than SIMPLE_LAND_DESC_SPEED_MIN_BELOW_M; linear in between [m/s].
SIMPLE_LAND_DESC_SPEED_MIN_MPS=0.1
# Below this height (world Z) the min descent speed is used [m].
SIMPLE_LAND_DESC_SPEED_MIN_BELOW_M=0.7
# Below this height above world Z=0, switch to the final push-down [m].
SIMPLE_LAND_FINAL_HEIGHT_M=0.25
# In the final push-down the setpoint is put this far below Z=0, so the drone keeps pressing down until PX4 detects the landing [m].
SIMPLE_LAND_PUSH_DOWN_M=10.0
